PKG_INVOICING                                                                                                                    PACKAGE BODY          5     PROCEDURE compute_preview(p_tenant_id IN VARCHAR2, p_period_start IN DATE,
PKG_INVOICING                                                                                                                    PACKAGE BODY          6                               p_period_end IN DATE) IS
PKG_INVOICING                                                                                                                    PACKAGE BODY          7         v_exempt CHAR(1) := 'N';
PKG_INVOICING                                                                                                                    PACKAGE BODY          8     BEGIN
PKG_INVOICING                                                                                                                    PACKAGE BODY          9         g_plan_code := NULL;
PKG_INVOICING                                                                                                                    PACKAGE BODY         10         g_plan_fee := NULL;
PKG_INVOICING                                                                                                                    PACKAGE BODY         11         BEGIN
PKG_INVOICING                                                                                                                    PACKAGE BODY         12             SELECT code, monthly_fee INTO g_plan_code, g_plan_fee
PKG_INVOICING                                                                                                                    PACKAGE BODY         13               FROM (SELECT p.code, p.monthly_fee
PKG_INVOICING                                                                                                                    PACKAGE BODY         14                       FROM subscriptions s, plans p
PKG_INVOICING                                                                                                                    PACKAGE BODY         15                      WHERE p.id = s.plan_id
PKG_INVOICING                                                                                                                    PACKAGE BODY         16                        AND s.tenant_id = p_tenant_id
PKG_INVOICING                                                                                                                    PACKAGE BODY         17                        AND s.starts_on <= p_period_end
PKG_INVOICING                                                                                                                    PACKAGE BODY         18                        AND (s.ends_on IS NULL OR s.ends_on >= p_period_start)
PKG_INVOICING                                                                                                                    PACKAGE BODY         19                      ORDER BY s.starts_on DESC)
PKG_INVOICING                                                                                                                    PACKAGE BODY         20              WHERE ROWNUM <= 1;
PKG_INVOICING                                                                                                                    PACKAGE BODY         21         EXCEPTION
PKG_INVOICING                                                                                                                    PACKAGE BODY         22             WHEN NO_DATA_FOUND THEN NULL;
PKG_INVOICING                                                                                                                    PACKAGE BODY         23         END;
PKG_INVOICING                                                                                                                    PACKAGE BODY         24
PKG_INVOICING                                                                                                                    PACKAGE BODY         25         pkg_rating.compute_rating(p_tenant_id, p_period_start, p_period_end);
PKG_INVOICING                                                                                                                    PACKAGE BODY         26         g_overage := pkg_rating.g_overage_amount;
PKG_INVOICING                                                                                                                    PACKAGE BODY         27
PKG_INVOICING                                                                                                                    PACKAGE BODY         28         -- Sum open credit notes one row at a time.
PKG_INVOICING                                                                                                                    PACKAGE BODY         29         g_credit := 0;
PKG_INVOICING                                                                                                                    PACKAGE BODY         30         FOR r IN (SELECT remaining_amount FROM credit_notes
PKG_INVOICING                                                                                                                    PACKAGE BODY         31                    WHERE tenant_id = p_tenant_id AND remaining_amount > 0) LOOP
PKG_INVOICING                                                                                                                    PACKAGE BODY         32             g_credit := g_credit + NVL(r.remaining_amount, 0);
PKG_INVOICING                                                                                                                    PACKAGE BODY         33         END LOOP;
PKG_INVOICING                                                                                                                    PACKAGE BODY         34
PKG_INVOICING                                                                                                                    PACKAGE BODY         35         BEGIN
PKG_INVOICING                                                                                                                    PACKAGE BODY         36             SELECT NVL(tax_exempt_yn, 'N') INTO v_exempt
PKG_INVOICING                                                                                                                    PACKAGE BODY         37               FROM tenants WHERE id = p_tenant_id;
PKG_INVOICING                                                                                                                    PACKAGE BODY         38         EXCEPTION
PKG_INVOICING                                                                                                                    PACKAGE BODY         39             WHEN NO_DATA_FOUND THEN v_exempt := 'N';
PKG_INVOICING                                                                                                                    PACKAGE BODY         40         END;
PKG_INVOICING                                                                                                                    PACKAGE BODY         41         -- NULL plan fee / overage propagates to NULL tax, exactly like the
PKG_INVOICING                                                                                                                    PACKAGE BODY         42         -- Postgres original's (v_plan.monthly_fee + overage) * 0.0825.
PKG_INVOICING                                                                                                                    PACKAGE BODY         43         SELECT DECODE(v_exempt, 'Y', 0,
PKG_INVOICING                                                                                                                    PACKAGE BODY         44                       (g_plan_fee + g_overage) * TAX_RATE)
PKG_INVOICING                                                                                                                    PACKAGE BODY         45           INTO g_tax FROM dual;
PKG_INVOICING                                                                                                                    PACKAGE BODY         46     END compute_preview;

PKG_INVOICING                                                                                                                    PACKAGE BODY         48     FUNCTION fn_invoice_preview(p_tenant_id IN VARCHAR2, p_period_start IN DATE,
PKG_INVOICING                                                                                                                    PACKAGE BODY         49                                 p_period_end IN DATE) RETURN SYS_REFCURSOR IS
PKG_INVOICING                                                                                                                    PACKAGE BODY         50         v_cur        SYS_REFCURSOR;
PKG_INVOICING                                                                                                                    PACKAGE BODY         51         v_credit_app NUMBER;
PKG_INVOICING                                                                                                                    PACKAGE BODY         52         v_charge_cap NUMBER;
PKG_INVOICING                                                                                                                    PACKAGE BODY         53     BEGIN
PKG_INVOICING                                                                                                                    PACKAGE BODY         54         compute_preview(p_tenant_id, p_period_start, p_period_end);
PKG_INVOICING                                                                                                                    PACKAGE BODY         55         -- Postgres LEAST ignores NULL arguments; Oracle propagates them.
PKG_INVOICING                                                                                                                    PACKAGE BODY         56         -- Collapse a NULL cap to g_credit so a no-plan period still applies
PKG_INVOICING                                                                                                                    PACKAGE BODY         57         -- the full credit balance, matching the original.
PKG_INVOICING                                                                                                                    PACKAGE BODY         58         v_charge_cap := ROUND(g_plan_fee + g_overage + g_tax, 2);
PKG_INVOICING                                                                                                                    PACKAGE BODY         59         v_credit_app := LEAST(g_credit, NVL(v_charge_cap, g_credit));
PKG_INVOICING                                                                                                                    PACKAGE BODY         60         OPEN v_cur FOR
PKG_INVOICING                                                                                                                    PACKAGE BODY         61             SELECT 1 AS line_no, 'plan' AS line_type, g_plan_code AS description,
PKG_INVOICING                                                                                                                    PACKAGE BODY         62                    ROUND(g_plan_fee, 2) AS amount, 0 AS tax_amount,
PKG_INVOICING                                                                                                                    PACKAGE BODY         63                    0 AS credit_applied, ROUND(g_plan_fee, 2) AS total
PKG_INVOICING                                                                                                                    PACKAGE BODY         64               FROM dual
PKG_INVOICING                                                                                                                    PACKAGE BODY         65             UNION ALL
PKG_INVOICING                                                                                                                    PACKAGE BODY         66             SELECT 2, 'usage', 'usage overage', ROUND(g_overage, 2), 0, 0,
PKG_INVOICING                                                                                                                    PACKAGE BODY         67                    ROUND(g_overage, 2)
PKG_INVOICING                                                                                                                    PACKAGE BODY         68               FROM dual
PKG_INVOICING                                                                                                                    PACKAGE BODY         69             UNION ALL
PKG_INVOICING                                                                                                                    PACKAGE BODY         70             SELECT 3, 'tax', 'regional tax', g_tax / 2, 0, 0, g_tax / 2 FROM dual
PKG_INVOICING                                                                                                                    PACKAGE BODY         71             UNION ALL
PKG_INVOICING                                                                                                                    PACKAGE BODY         72             SELECT 4, 'tax', 'local tax', g_tax / 2, 0, 0, g_tax / 2 FROM dual
PKG_INVOICING                                                                                                                    PACKAGE BODY         73             UNION ALL
PKG_INVOICING                                                                                                                    PACKAGE BODY         74             SELECT 5, 'credit', 'credit notes', 0, 0, v_credit_app, -v_credit_app
PKG_INVOICING                                                                                                                    PACKAGE BODY         75               FROM dual;
PKG_INVOICING                                                                                                                    PACKAGE BODY         76         RETURN v_cur;
PKG_INVOICING                                                                                                                    PACKAGE BODY         77     END fn_invoice_preview;

PKG_RATING                                                                                                                       PACKAGE BODY         35         EXCEPTION
PKG_RATING                                                                                                                       PACKAGE BODY         36             WHEN NO_DATA_FOUND THEN NULL;
PKG_RATING                                                                                                                       PACKAGE BODY         37         END;
PKG_RATING                                                                                                                       PACKAGE BODY         38
PKG_RATING                                                                                                                       PACKAGE BODY         39         BEGIN
PKG_RATING                                                                                                                       PACKAGE BODY         40             SELECT included_units, overage_rate INTO v_included, v_rate
PKG_RATING                                                                                                                       PACKAGE BODY         41               FROM plans WHERE id = v_plan_id;
PKG_RATING                                                                                                                       PACKAGE BODY         42         EXCEPTION
PKG_RATING                                                                                                                       PACKAGE BODY         43             WHEN NO_DATA_FOUND THEN NULL;
PKG_RATING                                                                                                                       PACKAGE BODY         44         END;
PKG_RATING                                                                                                                       PACKAGE BODY         45
PKG_RATING                                                                                                                       PACKAGE BODY         46         -- Sum the period's usage with a cursor loop, one row at a time,
PKG_RATING                                                                                                                       PACKAGE BODY         47         -- string-comparing the event date the mainframe way.
PKG_RATING                                                                                                                       PACKAGE BODY         48         FOR r IN (SELECT u.units, u.occurred_at FROM usage_events u
PKG_RATING                                                                                                                       PACKAGE BODY         49                    WHERE u.tenant_id = p_tenant_id) LOOP
PKG_RATING                                                                                                                       PACKAGE BODY         50             IF TO_CHAR(r.occurred_at, 'YYYYMMDD') >= TO_CHAR(p_period_start, 'YYYYMMDD')
PKG_RATING                                                                                                                       PACKAGE BODY         51                AND TO_CHAR(r.occurred_at, 'YYYYMMDD') <= TO_CHAR(p_period_end, 'YYYYMMDD') THEN
PKG_RATING                                                                                                                       PACKAGE BODY         52                 g_used_units := g_used_units + NVL(r.units, 0);
PKG_RATING                                                                                                                       PACKAGE BODY         53             END IF;
PKG_RATING                                                                                                                       PACKAGE BODY         54         END LOOP;
PKG_RATING                                                                                                                       PACKAGE BODY         55
PKG_RATING                                                                                                                       PACKAGE BODY         56         -- Rollover credit: prior three months of banked units, capped twice
PKG_RATING                                                                                                                       PACKAGE BODY         57         -- (once in the query, once right after, both to the same number).
PKG_RATING                                                                                                                       PACKAGE BODY         58         FOR r IN (SELECT rr.rollover_units
PKG_RATING                                                                                                                       PACKAGE BODY         59                     FROM rating_results rr, rating_periods rp
PKG_RATING                                                                                                                       PACKAGE BODY         60                    WHERE rp.id = rr.period_id
PKG_RATING                                                                                                                       PACKAGE BODY         61                      AND rp.tenant_id = p_tenant_id
PKG_RATING                                                                                                                       PACKAGE BODY         62                      AND rp.period_start < p_period_start
PKG_RATING                                                                                                                       PACKAGE BODY         63                      AND rp.period_start >= ADD_MONTHS(p_period_start, -3)) LOOP
PKG_RATING                                                                                                                       PACKAGE BODY         64             v_prior := v_prior + NVL(r.rollover_units, 0);
PKG_RATING                                                                                                                       PACKAGE BODY         65         END LOOP;
PKG_RATING                                                                                                                       PACKAGE BODY         66         -- Postgres LEAST/GREATEST ignore NULL arguments; Oracle propagates
PKG_RATING                                                                                                                       PACKAGE BODY         67         -- them. Collapse NULL operands to the other argument so a period with
PKG_RATING                                                                                                                       PACKAGE BODY         68         -- no covering plan rates the same as the original.
PKG_RATING                                                                                                                       PACKAGE BODY         69         v_prior := LEAST(NVL(2 * v_included, v_prior), v_prior);
PKG_RATING                                                                                                                       PACKAGE BODY         70
PKG_RATING                                                                                                                       PACKAGE BODY         71         g_quota_units := v_included;
PKG_RATING                                                                                                                       PACKAGE BODY         72         g_rollover_units := LEAST(v_prior, NVL(v_included * 2, v_prior));
PKG_RATING                                                                                                                       PACKAGE BODY         73         g_billable_units := GREATEST(NVL(g_used_units - g_rollover_units - v_included, 0), 0);
PKG_RATING                                                                                                                       PACKAGE BODY         74         -- Tier break at 101 units. Why 101? Nobody remembers.
PKG_RATING                                                                                                                       PACKAGE BODY         75         g_first_tier := LEAST(g_billable_units, 101);
PKG_RATING                                                                                                                       PACKAGE BODY         76         g_second_tier := GREATEST(g_billable_units - 101, 0);
PKG_RATING                                                                                                                       PACKAGE BODY         77         g_overage_amount := ROUND(g_first_tier * v_rate +
PKG_RATING                                                                                                                       PACKAGE BODY         78                                   g_second_tier * v_rate * 1.5, 2);
PKG_RATING                                                                                                                       PACKAGE BODY         79
PKG_RATING                                                                                                                       PACKAGE BODY         80         IF v_sub_status = 20 AND v_suspended_on IS NOT NULL
PKG_RATING                                                                                                                       PACKAGE BODY         81            AND v_suspended_on BETWEEN p_period_start AND p_period_end THEN
PKG_RATING                                                                                                                       PACKAGE BODY         82             v_factor := (p_period_end - v_suspended_on + 1) /
PKG_RATING                                                                                                                       PACKAGE BODY         83                         (p_period_end - p_period_start + 1);
PKG_RATING                                                                                                                       PACKAGE BODY         84             g_billable_units := ROUND(g_billable_units * v_factor);
PKG_RATING                                                                                                                       PACKAGE BODY         85             g_overage_amount := ROUND(g_overage_amount * v_factor, 2);
PKG_RATING                                                                                                                       PACKAGE BODY         86         END IF;
PKG_RATING                                                                                                                       PACKAGE BODY         87
PKG_RATING                                                                                                                       PACKAGE BODY         88         pkg_ow_util.log_msg('RATING', 'compute tenant=' || p_tenant_id ||
PKG_RATING                                                                                                                       PACKAGE BODY         89             ' used=' || TO_CHAR(NVL(g_used_units, -1)) ||
PKG_RATING                                                                                                                       PACKAGE BODY         90             ' billable=' || TO_CHAR(NVL(g_billable_units, -1)));
PKG_RATING                                                                                                                       PACKAGE BODY         91     END compute_rating;

PKG_RATING                                                                                                                       PACKAGE BODY        133     PROCEDURE sp_finalize_rating(p_tenant_id IN VARCHAR2, p_period_start IN DATE,
PKG_RATING                                                                                                                       PACKAGE BODY        134                                  p_period_end IN DATE) IS
PKG_RATING                                                                                                                       PACKAGE BODY        135         v_period_id VARCHAR2(36);
PKG_RATING                                                                                                                       PACKAGE BODY        136         v_result_id VARCHAR2(36);
PKG_RATING                                                                                                                       PACKAGE BODY        137         v_sub_id    VARCHAR2(36);
PKG_RATING                                                                                                                       PACKAGE BODY        138     BEGIN
PKG_RATING                                                                                                                       PACKAGE BODY        139         v_period_id := pkg_ow_util.f_md5_uuid(
PKG_RATING                                                                                                                       PACKAGE BODY        140             p_tenant_id || TO_CHAR(p_period_start, 'YYYY-MM-DD'));
PKG_RATING                                                                                                                       PACKAGE BODY        141
PKG_RATING                                                                                                                       PACKAGE BODY        142         BEGIN
PKG_RATING                                                                                                                       PACKAGE BODY        143             SELECT id INTO v_sub_id
PKG_RATING                                                                                                                       PACKAGE BODY        144               FROM (SELECT s.id FROM subscriptions s
PKG_RATING                                                                                                                       PACKAGE BODY        145                      WHERE s.tenant_id = p_tenant_id
PKG_RATING                                                                                                                       PACKAGE BODY        146                        AND s.starts_on <= p_period_end
PKG_RATING                                                                                                                       PACKAGE BODY        147                        AND (s.ends_on IS NULL OR s.ends_on >= p_period_start)
PKG_RATING                                                                                                                       PACKAGE BODY        148                      ORDER BY s.starts_on DESC)
PKG_RATING                                                                                                                       PACKAGE BODY        149              WHERE ROWNUM <= 1;
PKG_RATING                                                                                                                       PACKAGE BODY        150         EXCEPTION
PKG_RATING                                                                                                                       PACKAGE BODY        151             WHEN NO_DATA_FOUND THEN v_sub_id := NULL;
PKG_RATING                                                                                                                       PACKAGE BODY        152         END;
PKG_RATING                                                                                                                       PACKAGE BODY        153
PKG_RATING                                                                                                                       PACKAGE BODY        154         -- Upsert by trying the INSERT and falling back to UPDATE.
PKG_RATING                                                                                                                       PACKAGE BODY        155         BEGIN
PKG_RATING                                                                                                                       PACKAGE BODY        156             INSERT INTO rating_periods (id, tenant_id, period_start, period_end)
PKG_RATING                                                                                                                       PACKAGE BODY        157             VALUES (v_period_id, p_tenant_id, p_period_start, p_period_end);
PKG_RATING                                                                                                                       PACKAGE BODY        158         EXCEPTION
PKG_RATING                                                                                                                       PACKAGE BODY        159             WHEN DUP_VAL_ON_INDEX THEN
PKG_RATING                                                                                                                       PACKAGE BODY        160                 UPDATE rating_periods
PKG_RATING                                                                                                                       PACKAGE BODY        161                    SET period_end = p_period_end
PKG_RATING                                                                                                                       PACKAGE BODY        162                  WHERE tenant_id = p_tenant_id
PKG_RATING                                                                                                                       PACKAGE BODY        163                    AND period_start = p_period_start;
PKG_RATING                                                                                                                       PACKAGE BODY        164         END;
PKG_RATING                                                                                                                       PACKAGE BODY        165
PKG_RATING                                                                                                                       PACKAGE BODY        166         compute_rating(p_tenant_id, p_period_start, p_period_end);
PKG_RATING                                                                                                                       PACKAGE BODY        167
PKG_RATING                                                                                                                       PACKAGE BODY        168         v_result_id := pkg_ow_util.f_md5_uuid(v_period_id);
PKG_RATING                                                                                                                       PACKAGE BODY        169         BEGIN
PKG_RATING                                                                                                                       PACKAGE BODY        170             INSERT INTO rating_results (
PKG_RATING                                                                                                                       PACKAGE BODY        171                 id, period_id, subscription_id, used_units, quota_units,
PKG_RATING                                                                                                                       PACKAGE BODY        172                 rollover_units, billable_units, overage_amount, created_at
PKG_RATING                                                                                                                       PACKAGE BODY        173             ) VALUES (
PKG_RATING                                                                                                                       PACKAGE BODY        174                 v_result_id, v_period_id, v_sub_id, g_used_units, g_quota_units,
PKG_RATING                                                                                                                       PACKAGE BODY        175                 GREATEST(g_quota_units - g_used_units, 0),
PKG_RATING                                                                                                                       PACKAGE BODY        176                 g_billable_units, g_overage_amount, CAST(p_period_end AS TIMESTAMP)
PKG_RATING                                                                                                                       PACKAGE BODY        177             );
PKG_RATING                                                                                                                       PACKAGE BODY        178         EXCEPTION
PKG_RATING                                                                                                                       PACKAGE BODY        179             WHEN DUP_VAL_ON_INDEX THEN
PKG_RATING                                                                                                                       PACKAGE BODY        180                 UPDATE rating_results
PKG_RATING                                                                                                                       PACKAGE BODY        181                    SET used_units = g_used_units,
PKG_RATING                                                                                                                       PACKAGE BODY        182                        rollover_units = GREATEST(g_quota_units - g_used_units, 0),
PKG_RATING                                                                                                                       PACKAGE BODY        183                        billable_units = g_billable_units,
PKG_RATING                                                                                                                       PACKAGE BODY        184                        overage_amount = g_overage_amount
PKG_RATING                                                                                                                       PACKAGE BODY        185                  WHERE id = v_result_id;
PKG_RATING                                                                                                                       PACKAGE BODY        186         END;
PKG_RATING                                                                                                                       PACKAGE BODY        187         pkg_ow_util.log_msg('RATING', 'finalized period=' || v_period_id);
PKG_RATING                                                                                                                       PACKAGE BODY        188     END sp_finalize_rating;
PKG_RATING                                                                                                                       PACKAGE BODY        189
PKG_RATING                                                                                                                       PACKAGE BODY        190 END pkg_rating;
